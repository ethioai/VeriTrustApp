        <h1 class="text-3xl font-black text-indigo-400 mb-2">VeriTrust</h1>

        <!-- The Scanner Window -->
        <div class="relative overflow-hidden rounded-2xl bg-black border-4 border-gray-700 mb-6 aspect-square">
            <video id="v" class="absolute w-full h-full object-cover"></video>
            <div class="absolute inset-0 border-2 border-white/20 m-12 pointer-events-none rounded-xl border-dashed"></div>
        </div>

        <!-- The Result Panel -->
        <div id="res" class="hidden animate-pulse">
            <div id="res-box" class="p-4 rounded-xl border-2 flex flex-col items-center">
                <p id="res-status" class="text-xl font-black uppercase mb-2"></p>
                <div id="res-details" class="text-sm space-y-1 text-center"></div>
            </div>
        </div>

        <div id="idle-text" class="text-center text-gray-500 animate-pulse text-sm">
            Scanning for certificate QR...
        </div>
    </div>

    <script>
        const v = document.getElementById('v');
        const res = document.getElementById('res');
        const resBox = document.getElementById('res-box');
        const resStatus = document.getElementById('res-status');
        const resDetails = document.getElementById('res-details');
        const idleText = document.getElementById('idle-text');

        // Access Web Cam
        navigator.mediaDevices.getUserMedia({ video: { facingMode: "environment" } })
            .then(s => { v.srcObject = s; v.play(); requestAnimationFrame(tick); });

        function tick() {
            if (v.readyState === v.HAVE_ENOUGH_DATA) {
                const canvas = document.createElement('canvas');
                canvas.width = v.videoWidth; canvas.height = v.videoHeight;
                const ctx = canvas.getContext('2d');
                ctx.drawImage(v, 0, 0);
                const imageData = ctx.getImageData(0,0,canvas.width,canvas.height);
                const code = jsQR(imageData.data, imageData.width, imageData.height);

                if (code) { 
                    try { verify(code.data); } catch(e) { console.error("Verify Err:", e); }
                }
            }
            requestAnimationFrame(tick);
        }

        function verify(payload) {
            try {
                // 1. Unpack Envelope
                const env = JSON.parse(atob(payload));
                
                // 2. Deterministic Key Sort (Must match Python Canonical JSON exactly)
                const canonicalString = JSON.stringify(env.d, Object.keys(env.d).sort());
                const msgUint8 = new TextEncoder().encode(canonicalString);
                
                // 3. Convert signature and public key from Base64/Hex to Bytes
                const sigUint8 = Uint8Array.from(atob(env.s), c => c.charCodeAt(0));
                const pubKeyUint8 = new Uint8Array(env.p.match(/.{1,2}/g).map(b => parseInt(b, 16)));

                // 4. Ed25519 Math Proof
                const isValid = nacl.sign.detached.verify(msgUint8, sigUint8, pubKeyUint8);

                updateUI(isValid, env.d, env.i);
            } catch (err) {
                console.error("Format error:", err);
            }
        }

        function updateUI(valid, data, issuer) {
            idleText.classList.add('hidden');
            res.classList.remove('hidden');
            
            if (valid) {
                resBox.className = "p-4 rounded-xl border-2 border-green-500 bg-green-500/10 text-green-400 flex flex-col items-center";
                resStatus.innerText = "✓ Authentic";
                resDetails.innerHTML = `
                    <p class='font-bold text-lg'>${data.name}</p>
                    <p>${data.major}</p>
                    <p>Class of ${data.year} | GPA: ${data.gpa.toFixed(2)}</p>
                    <p class='text-[10px] mt-2 opacity-50 uppercase tracking-tighter'>Issuer: ${issuer}</p>
                `;
            } else {
                resBox.className = "p-4 rounded-xl border-2 border-red-500 bg-red-500/10 text-red-400 flex flex-col items-center";
                resStatus.innerText = "❌ Tampered Card";
                resDetails.innerText = "The signature failed. Data is modified.";
            }
            
            // Clear result after 3 seconds
            setTimeout(() => {
                res.classList.add('hidden');
                idleText.classList.remove('hidden');
            }, 3000);
        }
    </script>
